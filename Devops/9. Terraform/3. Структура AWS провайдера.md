```
terraform/
│
├── modules/                              # Чистые переиспользуемые Terraform-модули
│   │
│   ├── vpc/                              # Модуль сети
│   │   ├── main.tf                       # VPC, subnets, routes, NAT и т.д.
│   │   ├── variables.tf                  # Входные параметры модуля
│   │   ├── outputs.tf                    # vpc_id, subnet_ids и т.д.
│   │   └── versions.tf                   # Terraform + provider versions
│   │
│   ├── app/                              # Инфраструктура приложения
│   │   ├── main.tf                       # LB/ECS/EKS/IAM и другие ресурсы app
│   │   ├── variables.tf                  # image, replicas, ports и т.д.
│   │   ├── outputs.tf                    # app_url, service_id и т.д.
│   │   └── versions.tf                   # Версии Terraform/providers
│   │
│   ├── postgres/                         # Модуль PostgreSQL
│   │   ├── main.tf                       # RDS/PostgreSQL
│   │   ├── variables.tf                  # size, disk, version и т.д.
│   │   ├── outputs.tf                    # endpoint, port
│   │   └── versions.tf
│   │
│   └── s3/                               # Модуль Object Storage/S3
│       ├── main.tf
│       ├── variables.tf
│       ├── outputs.tf
│       └── versions.tf
│
│
└── terragrunt/                           # LIVE-конфигурация: где и с какими параметрами запускать modules
    │
    ├── terragrunt.hcl                    # Общие настройки для ВСЕХ env:
    │                                     # remote state, provider, общие locals
    │
    ├── dev/                              # DEV environment
    │   │
    │   ├── env.hcl                       # environment="dev", account_id, domain и т.п.
    │   │
    │   ├── vpc/
    │   │   └── terragrunt.hcl            # Вызывает modules/vpc + DEV параметры
    │   │
    │   ├── app/
    │   │   └── terragrunt.hcl            # Вызывает modules/app + DEV параметры
    │   │
    │   ├── postgres/
    │   │   └── terragrunt.hcl            # Вызывает modules/postgres + DEV параметры
    │   │
    │   └── s3/
    │       └── terragrunt.hcl            # Вызывает modules/s3 + DEV параметры
    │
    ├── stage/                            # STAGE environment
    │   │
    │   ├── env.hcl                       # environment="stage", свои значения
    │   │
    │   ├── vpc/
    │   │   └── terragrunt.hcl            # modules/vpc для stage
    │   │
    │   ├── app/
    │   │   └── terragrunt.hcl            # modules/app для stage
    │   │
    │   ├── postgres/
    │   │   └── terragrunt.hcl            # modules/postgres для stage
    │   │
    │   └── s3/
    │       └── terragrunt.hcl            # modules/s3 для stage
    │
    └── prod/                             # PROD environment
        │
        ├── env.hcl                       # environment="prod", prod account/domain и т.д.
        │
        ├── vpc/
        │   └── terragrunt.hcl            # modules/vpc для production
        │
        ├── app/
        │   └── terragrunt.hcl            # modules/app для production
        │
        ├── postgres/
        │   └── terragrunt.hcl            # modules/postgres для production
        │
        └── s3/
            └── terragrunt.hcl            # modules/s3 для production
```
