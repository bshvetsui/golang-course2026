## основные команды для работы 
```
terraform plan       # посмотреть изменения что именно поменяется не применяя
terraform apply      # применить изменения
terraform destroy    # удалить инфраструктуру
```

## Примеры работы 

### 1. Добавили новый сервис `reports`
```hcl
# services.tf

module "reports" { # НОВОЕ: подключили девятый сервис
  source = "../../modules/service"

  name        = "reports"
  namespace   = "app"
  environment = var.environment

  cpu    = 500
  memory = 512
}
```

### 2. Увеличили БД `orders`
```hcl
# postgres.tf

module "orders_db" {
  source = "../../modules/postgres"

  name        = "orders"
  environment = var.environment

  cpu       = 4   # ИЗМЕНИЛИ: было 2
  memory_gb = 16  # ИЗМЕНИЛИ: было 8
  disk_gb   = 200 # ИЗМЕНИЛИ: было 100
}
```

### 3. Добавили БД для нового `reports`
```hcl
# postgres.tf

module "reports_db" { # НОВОЕ: отдельная БД для reports
  source = "../../modules/postgres"

  name          = "reports"
  database_name = "reports"
  environment   = var.environment

  disk_gb = 50
}
```

### 4. Добавили Kafka topic
```hcl
# kafka.tf

module "orders_events" { # НОВОЕ: topic для событий orders
  source = "../../modules/kafka-topic"

  name = "orders.events"

  partitions         = 12
  replication_factor = 3
  retention_ms       = 604800000
}
```

### 5. Поменяли домен фронта
```hcl
# frontend.tf

module "frontend" {
  source = "../../modules/frontend"

  name        = "app"
  environment = var.environment

  domain      = "new.example.com" # ИЗМЕНИЛИ: старый домен заменили
  backend_url = module.api.endpoint
}
```
