Лучший способ выучить синтаксис - практика. Достаточно прочитать пример 1 раз. 
```
stages:
  - build        # TypeScript compile + production bundle
  - types        # отдельная проверка типов без генерации файлов
  - lint         # ESLint
  - component    # component tests
  - unit         # unit tests
  - package      # Docker + Helm
  - publish      # push в Harbor/Nexus


variables:
  PNPM_STORE_DIR: ".pnpm-store"                          # директория кеша pnpm
  IMAGE_NAME: "frontend/web"                             # имя Docker image
  CHART_NAME: "frontend"                                 # имя Helm chart
  HARBOR_HOST: "harbor.company.local"                    # адрес registry
  HARBOR_PROJECT: "frontend"                             # project/repository namespace в Harbor


.frontend:
  image: node:22-alpine                                  # Node environment для frontend jobs

  before_script:
    - corepack enable                             # включить pnpm (пакетный менеджер) + Corepack (утилита для версий)
    - pnpm config set store-dir "$PNPM_STORE_DIR"        # использовать локальный pnpm cache
    - pnpm install --frozen-lockfile                     # установить зависимости строго по pnpm-lock.yaml

  cache:
    key:
      files:
        - pnpm-lock.yaml                                 # новый lockfile (файл фиксирующий версии пакетов) → новый cache key
    paths:
      - .pnpm-store/                                     # кешировать скачанные pnpm packages (эти пакеты надо фиксировать)


build:
  extends: .frontend
  stage: build

  script:
    - rm -rf dist                                        # удалить предыдущий build
    - pnpm exec tsc -b                                   # проверка типов
    - pnpm exec vite build                               # vite "компилятор кода", билдим в папку → dist/
    - ls -lah dist                                       # посмотреть содержимое production build
    - du -sh dist                                        # посмотреть размер bundle (сбилженного)

  artifacts:
    paths:
      - dist/                                            # передать готовый frontend в Docker job
    expire_in: 1 day                                     # удалить artifact через сутки


types:
  extends: .frontend
  stage: types

  script:
    - pnpm exec tsc --noEmit                             # только type checking, без генерации JS


lint:
  extends: .frontend
  stage: lint

  script:
    - >
      pnpm exec eslint src                               # проверить исходники ESLint
      --ext .ts,.tsx                                     # проверять TypeScript и React TSX
      --max-warnings=0                                   # warning тоже делает job failed


component:
  extends: .frontend
  stage: component

  script:
    - >
      pnpm exec vitest run                               # запустить Vitest один раз в CI
      --config vitest.component.config.ts                # конфиг именно component tests


unit:
  extends: .frontend
  stage: unit

  script:
    - >
      pnpm exec vitest run                               # unit-тесты utils/hooks/services и т.п.
      --config vitest.unit.config.ts                     # отдельный unit test config
      --coverage                                         # собрать test coverage


helm-package:
  stage: package

  image:
    name: alpine/helm                                    # контейнер с Helm CLI
    entrypoint: [""]

  variables:
    VERSION: "0.0.${CI_PIPELINE_IID}"                    # версия chart из номера pipeline

  script:
    - helm lint deploy/helm/frontend                     # проверить Helm chart
    - >
      helm package deploy/helm/frontend                  # упаковать директорию chart
      --version "$VERSION"                               # версия Helm chart
      --app-version "$CI_COMMIT_SHA"                     # версия приложения = Git commit
      --destination ./packages                           # получить packages/frontend-x.x.x.tgz

  artifacts:
    paths:
      - packages/*.tgz                                   # передать Helm archive в publish


docker-image:
  stage: package

  image: docker:cli                                      # Docker CLI внутри job

  services:
    - docker:dind                                        # Docker daemon для сборки image

  needs:
    - job: build                                         # дождаться frontend build
      artifacts: true                                    # скачать dist/ из build job

  script:
    - >
      docker build                                       # собрать production image
      -t "$HARBOR_HOST/$HARBOR_PROJECT/$IMAGE_NAME:$CI_COMMIT_SHORT_SHA"
      .

    - >
      docker save                                        # сохранить image в tar между jobs
      -o frontend-image.tar
      "$HARBOR_HOST/$HARBOR_PROJECT/$IMAGE_NAME:$CI_COMMIT_SHORT_SHA"

  artifacts:
    paths:
      - frontend-image.tar                               # передать Docker image в publish job


publish:
  stage: publish

  image: docker:cli
  services:
    - docker:dind

  needs:
    - docker-image                                       # получить frontend-image.tar
    - helm-package                                       # получить frontend-x.x.x.tgz

  script:
    - >
      echo "$HARBOR_PASSWORD" |
      docker login "$HARBOR_HOST"                        # авторизация Docker в Harbor
      -u "$HARBOR_USER"
      --password-stdin

    - docker load -i frontend-image.tar                  # загрузить сохранённый image обратно в Docker

    - >
      docker push
      "$HARBOR_HOST/$HARBOR_PROJECT/$IMAGE_NAME:$CI_COMMIT_SHORT_SHA" # Docker image → Harbor

    - apk add --no-cache helm                            # добавить Helm CLI в docker:cli image

    - >
      echo "$HARBOR_PASSWORD" |
      helm registry login "$HARBOR_HOST"                 # авторизация Helm в OCI registry
      -u "$HARBOR_USER"
      --password-stdin

    - >
      helm push packages/*.tgz
      "oci://$HARBOR_HOST/$HARBOR_PROJECT"               # Helm .tgz → Harbor
```
