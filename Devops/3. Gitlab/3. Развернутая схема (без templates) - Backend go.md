Лучший способ выучить синтаксис - практика. Достаточно прочитать пример 1 раз. 
```
stages:
  - verify      # зависимости / go.mod / go.sum
  - lint        # статический анализ Go-кода
  - test        # unit + migration + integration
  - build       # компиляция production binary
  - package     # Docker image + Helm chart
  - publish     # push артефактов в Harbor/Nexus


verify:
  stage: verify
  script:
    - go mod verify                                      # проверить checksum зависимостей из go.sum
    - go mod tidy                                        # привести go.mod/go.sum в актуальное состояние
    - git diff --exit-code -- go.mod go.sum             # упасть, если tidy изменил файлы


lint:
  stage: lint
  script:
    - golangci-lint run ./...                            # govet/staticcheck/errcheck/unused и другие линтеры


unit:
  stage: test
  script:
    - go test -race -cover ./...                         # unit-тесты + data race detector + coverage


migration:
  stage: test
  script:
    - goose postgres "$DATABASE_URL" up                  # накатить все миграции на тестовую БД
    - goose postgres "$DATABASE_URL" down                # проверить rollback последней миграции
    - goose postgres "$DATABASE_URL" up                  # проверить, что после rollback она снова накатывается


integration:
  stage: test
  script:
    - goose postgres "$DATABASE_URL" up                  # подготовить актуальную схему тестовой БД
    - go test -tags=integration ./...                    # тесты с PostgreSQL/Redis/Kafka и т.п.


build:
  stage: build
  script:
    - >
      CGO_ENABLED=0                                      # собрать без зависимости от системных C-библиотек
      GOOS=linux                                         # binary предназначен для Linux-контейнера
      go build                                           # компиляция Go-кода + type checking
      -trimpath                                          # убрать локальные пути CI runner из binary
      -o bin/api                                         # выходной executable
      ./cmd/api                                          # main package приложения

  artifacts:
    paths:
      - bin/api                                          # передать готовый binary в следующие jobs


docker:
  stage: package
  script:
    - docker build -t "$IMAGE:$CI_COMMIT_SHORT_SHA" .    # собрать Docker image с конкретным commit tag


helm:
  stage: package
  script:
    - helm lint deploy/helm/api                          # проверить Chart.yaml/values/templates
    - helm package deploy/helm/api --destination packages/ # упаковать Helm chart в .tgz

  artifacts:
    paths:
      - packages/*.tgz                                   # передать Helm archive в publish


publish:
  stage: publish
  script:
    - docker push "$IMAGE:$CI_COMMIT_SHORT_SHA"          # Docker image → Harbor/Nexus
    - helm push packages/*.tgz "$HELM_REGISTRY"         # Helm .tgz → OCI registry
```
